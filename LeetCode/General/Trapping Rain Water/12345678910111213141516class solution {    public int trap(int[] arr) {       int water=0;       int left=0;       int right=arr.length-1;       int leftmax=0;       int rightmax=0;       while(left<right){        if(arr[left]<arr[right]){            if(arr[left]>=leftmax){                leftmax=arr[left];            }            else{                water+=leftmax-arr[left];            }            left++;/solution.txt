class Solution {
    public int trap(int[] arr) {
       int water=0;
       int left=0;
       int right=arr.length-1;
       int leftmax=0;
       int rightmax=0;
       while(left<right){
        if(arr[left]<arr[right]){
            if(arr[left]>=leftmax){
                leftmax=arr[left];
            }
            else{
                water+=leftmax-arr[left];
            }
            left++;
